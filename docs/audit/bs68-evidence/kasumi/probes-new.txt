use std.signal
use std.term
use std.env

fn group_row(g: int, meaning: int) -> str {
    os_signal_send_group(g, meaning) else |e| {
        return match e {
            unsupported => "unsupported",
            invalid => "invalid",
            not_found => "not_found",
            denied => "denied",
            io => "io",
        }
    }
    "ok"
}

fn main() -> !int {
    os_set_pgid(0)?
    let mine = term.group()?
    print("leads a group of its own: {mine == env.pid()}")
    print("its own group, nothing sent: {group_row(mine, 0)}")
    print("group 1: {group_row(1, 0)}")
    print("a group nobody leads: {group_row(2147483647, 0)}")
    print("two meanings at once: {group_row(mine, signal.user() | signal.alarm())}")
    print("host numbers past the table's: {os_signal_last()? > 28}")
    0
}

use std.signal

fn changed(kid: int) -> int ! {io} {
    while true {
        let (h, status) = os_wait_poll()?
        if h == kid { return status }
        let _ = signal.wait(signal.child())?
    }
    0
}

fn main() -> !int {
    signal.listen(signal.child())?
    var argv = List[str]()
    (mut argv).push("/bin/sleep")
    (mut argv).push("10")
    let kid = os_spawn(argv)?
    let pid = os_proc_pid(kid)?
    signal.send(pid, signal.stop())?
    let stopped = changed(kid)?
    print("a stop: {stopped >= 256}, by stop(): {stopped - 256 == signal.number(signal.stop())?}")
    signal.send(pid, signal.kill())?
    let ended = changed(kid)?
    print("an end: {ended < 0}, by kill(): {-ended == signal.number(signal.kill())?}")
    let (h, live) = os_wait_poll()?
    print("then: {h}, {live} live")
    0
}
